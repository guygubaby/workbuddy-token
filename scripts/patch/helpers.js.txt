// accountOf reads one account export: the {account:{uid,nickname}, auth:{
// accessToken, refreshToken, expiresAt, domain}} shape CodeBuddy's export
// tool writes, or a flat token object, camelCase or snake_case. Times are
// seconds in the export and kept as ms.
function accountOf(v) {
  if (!v || typeof v !== "object") throw new Error("holds no JSON object")
  const a = v.auth && typeof v.auth === "object" ? v.auth : v
  const one = (...xs) => {
    for (const x of xs) if (typeof x === "string" && x.trim()) return x.trim()
    return ""
  }
  const access = one(a.accessToken, a.access_token, a.access)
  const refresh = one(a.refreshToken, a.refresh_token, a.refresh)
  const uid = one(v.account?.uid, v.uid, a.uid)
  const nickname = one(v.account?.nickname, v.nickname, v.account?.name, a.nickname)
  const domain = one(a.domain, v.domain)
  const raw = Number(a.expiresAt ?? a.expires_at ?? a.expires ?? 0) || 0
  const expires = raw > 0 ? (raw < 1e12 ? raw * 1000 : raw) : 0
  if (!access) throw new Error("holds no accessToken")
  return { access, refresh, expires, uid, nickname, domain }
}

// importPath is a path as the user gave it: quotes and a leading ~ or
// file:// taken off.
function importPath(p) {
  let q = p.replace(/^['"]|['"]$/g, "").trim()
  if (q.startsWith("~/")) q = join(homedir(), q.slice(2))
  if (q.startsWith("file://")) q = new URL(q).pathname
  return q
}

// importAccounts reads what the account-file sign-in was given: a folder
// (every *.json in it, in name order), one or more paths on separate lines
// or separated by ";", a JSON array, or one account's JSON. Each file is
// read on its own: one that doesn't read is skipped with why, and the rest
// still import.
function importAccounts(input) {
  const text = typeof input === "string" ? input.trim() : ""
  if (!text) throw new Error("Paste the account JSON, or a folder or file path")
  const refs = []
  if (text.startsWith("{") || text.startsWith("[")) {
    let v
    try {
      v = JSON.parse(text)
    } catch {
      throw new Error("That isn't JSON")
    }
    ;(Array.isArray(v) ? v : [v]).forEach((x, i) => refs.push({ name: `#${i + 1}`, value: x }))
  } else {
    for (const p of text.split(/\r?\n|;/).map((x) => x.trim()).filter(Boolean)) {
      const q = importPath(p)
      let st
      try {
        st = statSync(q)
      } catch (e) {
        refs.push({ name: q, why: e?.code === "ENOENT" ? "no such file" : e?.message ?? String(e) })
        continue
      }
      if (!st.isDirectory()) {
        refs.push({ name: q, file: q })
        continue
      }
      let names
      try {
        names = readdirSync(q).filter((n) => n.toLowerCase().endsWith(".json")).sort()
      } catch (e) {
        refs.push({ name: q, why: e?.message ?? String(e) })
        continue
      }
      if (!names.length) refs.push({ name: q, why: "hold no .json file" })
      for (const n of names) refs.push({ name: join(q, n), file: join(q, n) })
    }
  }
  const accounts = []
  const skipped = []
  for (const r of refs) {
    if (r.why) {
      skipped.push({ name: r.name, why: r.why })
      continue
    }
    try {
      const v = r.file ? JSON.parse(readFileSync(r.file, "utf8")) : r.value
      accounts.push({ name: r.name, ...accountOf(v) })
    } catch (e) {
      skipped.push({ name: r.name, why: e?.message ?? String(e) })
    }
  }
  return { accounts, skipped }
}

// saveAccounts keeps each imported account as its own magpie sign-in: the
// first reaches magpie through the sign-in's own save, the rest through the
// host's client.auth.set. The key is the uid's hash, so importing the same
// account again updates it instead of adding it twice. OpenCode (no magpie)
// keeps only the one account its sign-in saves itself.
async function saveAccounts(site, client, magpie, accounts, skipped) {
  let saved = 0
  if (!magpie) return { saved, skipped }
  for (const a of accounts) {
    const key = site.id + "#" + createHash("sha1").update(a.uid || a.access).digest("hex").slice(0, 8)
    try {
      await client?.auth?.set?.({
        path: { id: key },
        body: {
          type: "oauth",
          access: a.access,
          refresh: a.refresh,
          expires: a.expires,
          accountId: a.nickname || a.uid || site.name,
          uid: a.uid,
          ...(a.domain ? { domain: a.domain } : {}),
          ...(a.nickname ? { nickname: a.nickname } : {}),
        },
      })
      saved++
    } catch (e) {
      skipped.push({ name: a.nickname || a.uid || "account", why: e?.message ?? String(e) })
    }
  }
  return { saved, skipped }
}

// liveImport is the upload page a sign-in left open, if any: a sign-in
// started again closes it first, so the port is free for the new page even
// when the first was never finished (magpie cancelled it, or the user just
// started another).
let liveImport = null

function stopImport(self, why) {
  if (!self) return Promise.resolve()
  if (liveImport === self) liveImport = null
  if (self.timer) clearTimeout(self.timer)
  if (why && !self.settled) {
    self.settled = true
    self.settle(why)
  }
  try {
    self.server?.closeAllConnections?.()
  } catch {}
  return new Promise((resolve) => {
    try {
      self.server?.close(() => resolve())
    } catch {
      resolve()
    }
  })
}

// browserImport serves that page for one sign-in and waits for it. magpie's
// sign-in prompts have no file picker, so the plugin brings its own: a
// one-time token in the URL, a port that maps into the container
// (WORKBUDDY_IMPORT_PORT, 3431 by default), and the address the browser
// reaches it at (WORKBUDDY_IMPORT_HOST, 127.0.0.1 by default).
async function browserImport(site, client, magpie) {
  const { createServer } = await import("node:http")
  // a page still open from an earlier sign-in holds the port: close it
  await stopImport(liveImport, { type: "failed", error: "Superseded by a newer account import" })
  const token = randomBytes(24).toString("hex")
  const wantPort = Number(process.env.WORKBUDDY_IMPORT_PORT || 3431)
  const publicHost = process.env.WORKBUDDY_IMPORT_HOST || "127.0.0.1"
  const bind = process.env.WORKBUDDY_IMPORT_BIND || (publicHost === "127.0.0.1" ? "127.0.0.1" : "0.0.0.0")
  const state = { server: null, settled: false, timer: null, settle: null }
  const finished = new Promise((r) => (state.settle = r))
  const server = createServer(async (req, res) => {
    const url = new URL(req.url ?? "/", "http://localhost")
    const say = (code, body, type = "text/plain; charset=utf-8") => {
      res.writeHead(code, { "content-type": type, "cache-control": "no-store", "x-content-type-options": "nosniff" })
      res.end(body)
    }
    if (url.searchParams.get("t") !== token) return say(404, "not found")
    if (req.method === "GET") return say(200, IMPORT_PAGE, "text/html; charset=utf-8")
    if (req.method !== "POST") return say(405, "method not allowed")
    if (state.settled) return say(409, "this import is already done")
    // the sign-in finishes only after this answer has left: its callback
    // closes the server, and closing it under the response cut the socket
    const answer = (result) => {
      const done = () => {
        if (state.settled) return
        state.settled = true
        state.settle(result)
      }
      res.once("finish", done)
      res.once("close", done)
    }
    let body = ""
    try {
      for await (const chunk of req) {
        body += chunk
        if (body.length > 4 << 20) throw new Error("the upload is over 4 MB")
      }
      const inb = JSON.parse(body)
      const files = Array.isArray(inb?.files) ? inb.files : [{ name: "pasted", text: String(inb?.text ?? "") }]
      const accounts = []
      const skipped = []
      files.forEach((f, i) => {
        try {
          const found = importAccounts(String(f?.text ?? ""))
          accounts.push(...found.accounts)
          skipped.push(...found.skipped)
        } catch (e) {
          skipped.push({ name: String(f?.name ?? `#${i + 1}`), why: e?.message ?? String(e) })
        }
      })
      if (!accounts.length) throw new Error(`No account imported${skipped[0] ? ` — ${skipped[0].name}: ${skipped[0].why}` : ""}`)
      const { saved } = await saveAccounts(site, client, magpie, accounts, skipped)
      const a = accounts[0]
      const result = {
        type: "success",
        access: a.access,
        refresh: a.refresh,
        expires: a.expires,
        accountId: a.nickname || a.uid || site.name,
        uid: a.uid,
        ...(a.domain ? { domain: a.domain } : {}),
        ...(a.nickname ? { nickname: a.nickname } : {}),
      }
      const said = `Imported ${saved} account${saved === 1 ? "" : "s"}${skipped.length ? `, skipped ${skipped.length}` : ""}`
      if (magpie && (accounts.length > 1 || skipped.length)) {
        try {
          await client?.tui?.showToast?.({ body: { title: site.name, message: said } })
        } catch {}
      }
      answer(result)
      return say(200, `${said}\n\n可以关闭此页面，回到 magpie。`)
    } catch (e) {
      answer({ type: "failed", error: e?.message ?? String(e) })
      return say(400, `没有导入账号：${e?.message ?? e}`)
    }
  })
  state.server = server
  try {
    await new Promise((resolve, reject) => {
      const onError = (e) => reject(e)
      server.once("error", onError)
      server.listen(wantPort, bind, () => {
        server.off("error", onError)
        resolve()
      })
    })
  } catch (e) {
    await stopImport(state)
    return { type: "failed", error: `Can't listen on ${bind}:${wantPort}: ${e?.message ?? e}` }
  }
  liveImport = state
  const port = server.address().port
  state.timer = setTimeout(() => {
    if (!state.settled) {
      state.settled = true
      state.settle({ type: "failed", error: "The import page timed out after 5 minutes" })
    }
    stopImport(state)
  }, 5 * 60 * 1000)
  return {
    url: `http://${publicHost}:${port}/import?t=${token}`,
    instructions: "Choose one or more account JSON files in the page that opens.",
    method: "auto",
    async callback() {
      try {
        return await finished
      } finally {
        await stopImport(state)
      }
    },
  }
}
